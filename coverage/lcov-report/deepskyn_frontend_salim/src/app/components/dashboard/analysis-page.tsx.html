                skinType: results.skinType || 'normal',

            />

        <AnimatePresence>
          {showAnalysisHistory && (

        <AnalysisCompareModal

          onClose={() => setCompareOpen(false)}
          allAnalyses={allAnalysesForCompare}
          initialA={analysis}
        />
      </div>
    </div>
