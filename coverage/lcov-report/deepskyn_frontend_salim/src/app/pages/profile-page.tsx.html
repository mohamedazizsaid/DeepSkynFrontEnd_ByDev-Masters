import React, { useState, useEffect } from 'react';
import { motion } from 'motion/react';
import { ArrowLeft, Loader2 } from 'lucide-react';
import { ProfileHero } from './components/profile-hero';
import { ProfileTabs, type Tab } from './components/profile-tabs';
import { PostComposer } from './components/post-composer';
import { PostCard } from './components/post-card';
import { AdCard } from './components/ad-card';
import { ProfileSidebar } from './components/profile-sidebar';
import { MOCK_POSTS } from './components/mock-posts';
import { MOCK_ADS, interleaveFeedWithAds } from './components/mock-ads';
import { useAuthStore } from '../../stores/auth.store';
import { useAccessibilityStore } from '../../stores/accessibility.store';
import toast from 'react-hot-toast';
 
export interface ProfilePageProps {
  onBack?: () => void;
  userId?: string; // If provided, show that user's profile instead of current user
}
 
export function ProfilePage({ onBack, userId }: ProfilePageProps) {
  const { user: currentUser } = useAuthStore();
  const { t } = useAccessibilityStore();
  const [activeTab, setActiveTab] = useState<Tab>('publications');
  const [isLoading, setIsLoading] = useState(false);
  const [posts, setPosts] = useState(MOCK_POSTS);
  const [feedWithAds, setFeedWithAds] = useState<any[]>([]);
  const [userProfile, setUserProfile] = useState({
    id: userId || currentUser?.sub || '',
    name: currentUser?.name || t('common.user'),
    email: currentUser?.email || 'user@example.com',
    avatar: currentUser?.avatar,
    coverPhoto: currentUser?.coverPhoto,
    bio: 'Passionnée par la skincare et le bien-être 💙 | Partage mes routines et conseils.',
    location: 'Paris, France',
    website: 'https://www.deepskyn.com',
    gender: currentUser?.gender,
    createdAt: currentUser?.createdAt || new Date().toISOString(),
    postsCount: 24,
    followersCount: 1250,
    followingCount: 342,
  });
 
  const isOwnProfile = !userId || userId === currentUser?.sub;
 
  // Initialize feed with ads
  useEffect(() => {
    const ads = MOCK_ADS.slice(0, 2); // Show 2 ads max in feed
    const interleaved = interleaveFeedWithAds(MOCK_POSTS, ads);
    setFeedWithAds(interleaved);
  }, []);
 
  const handleCoverPhotoUpload = async (url: string) => {
    setIsLoading(true);
    try {
      // Simulate API call
      await new Promise((resolve) => setTimeout(resolve, 1000));
      setUserProfile((prev) => ({ ...prev, coverPhoto: url }));
      // In real app, would call: await authService.updateProfile({ coverPhoto: url });
    } catch (error) {
      toast.error(t('common.error'));
      console.error(error);
    } finally {
      setIsLoading(false);
    }
  };
 
  const handleAvatarUpload = async (url: string) => {
    setIsLoading(true);
    try {
      // Simulate API call
      await new Promise((resolve) => setTimeout(resolve, 1000));
      setUserProfile((prev) => ({ ...prev, avatar: url }));
      // In real app, would call: await authService.updateProfile({ avatar: url });
    } catch (error) {
      toast.error(t('common.error'));
      console.error(error);
    } finally {
      setIsLoading(false);
    }
  };
 
  const handleCreatePost = async (content: string, media?: string) => {
    try {
      const newPost = {
        id: String(posts.length + 1),
        userId: userProfile.id,
        userName: userProfile.name,
        userAvatar: userProfile.avatar,
        content,
        media,
        createdAt: new Date().toISOString(),
        likes: 0,
        comments: 0,
        shares: 0,
        isLiked: false,
        isByCurrentUser: true,
      };
 
      setPosts((prev) => [newPost, ...prev]);
      setUserProfile((prev) => ({
        ...prev,
        postsCount: prev.postsCount + 1,
      }));
 
      // Regenerate feed with ads
      const ads = MOCK_ADS.slice(0, 2);
      const interleaved = interleaveFeedWithAds([newPost, ...posts], ads);
      setFeedWithAds(interleaved);
    } catch (error) {
      toast.error(t('common.error'));
      console.error(error);
    }
  };
 
  const handleDeletePost = (postId: string) => {
    setPosts((prev) => prev.filter((p) => p.id !== postId));
    setUserProfile((prev) => ({
      ...prev,
      postsCount: Math.max(0, prev.postsCount - 1),
    }));
 
    // Regenerate feed
    const newPosts = posts.filter((p) => p.id !== postId);
    const ads = MOCK_ADS.slice(0, 2);
    const interleaved = interleaveFeedWithAds(newPosts, ads);
    setFeedWithAds(interleaved);
  };
 
  const handleLikePost = (postId: string) => {
    // Local state only - no backend persistence
    setPosts((prev) =>
      prev.map((p) => (p.id === postId ? { ...p, isLiked: !p.isLiked } : p))
    );
  };
 
  return (
    <div className="min-h-screen relative" style={{ zIndex: 1 }}>
      {/* Header */}
      <div className="sticky top-0 z-50 bg-white dark:bg-slate-900 border-b border-[#E5E7EB] dark:border-slate-700 shadow-sm">
        <div className="max-w-6xl mx-auto px-6 py-4 flex items-center justify-between">
          <div className="flex items-center gap-4">
            {onBack && (
              <button
                onClick={onBack}
                className="flex items-center gap-2 text-[#6B7280] dark:text-gray-400 hover:text-[#0EA5E9] dark:hover:text-sky-300 transition-colors"
              >
                <ArrowLeft className="w-5 h-5" />
              </button>
            )}
            <h1 className="text-2xl font-bold text-[#111827] dark:text-gray-100">{t('community.profile')}</h1>
          </div>
        </div>
      </div>
 
      {/* Main Content */}
      <div className="max-w-6xl mx-auto px-6 py-8">
        {isLoading && (
          <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
            <Loader2 className="w-8 h-8 animate-spin text-white" />
          </div>
        )}
 
        {/* Profile Hero */}
        <ProfileHero
          user={userProfile}
          isOwnProfile={isOwnProfile}
          onEditProfile={() => toast(t('common.edit'))}
          onCoverPhotoUpload={handleCoverPhotoUpload}
          onAvatarUpload={handleAvatarUpload}
          onFollow={() => toast.success(t('common.follow'))}
        />
 
        {/* Tabs */}
        <ProfileTabs activeTab={activeTab} onTabChange={setActiveTab} />
 
        {/* Content Based on Active Tab */}
        {activeTab === 'publications' && (
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
            {/* Main Feed */}
            <div className="lg:col-span-2">
              {/* Post Composer - Only for own profile */}
              {isOwnProfile && (
                <PostComposer
                  userAvatar={userProfile.avatar}
                  userName={userProfile.name}
                  onSubmit={handleCreatePost}
                />
              )}
 
              {/* Feed */}
              {feedWithAds.length > 0 ? (
                <div>
                  {feedWithAds.map((item, idx) => (
                    <div key={idx}>
                      {item.type === 'post' ? (
                        <PostCard
                          post={item.data}
                          onLike={handleLikePost}
                          onDelete={handleDeletePost}
                        />
                      ) : (
                        <AdCard ad={item.data} />
                      )}
                    </div>
                  ))}
                </div>
              ) : (
                <div className="text-center py-12 bg-white dark:bg-slate-900 rounded-2xl border border-[#E5E7EB] dark:border-slate-700">
                  <p className="text-[#6B7280] dark:text-gray-400">{t('community.noPosts')}</p>
                </div>
              )}
            </div>
 
            {/* Sidebar */}
            <div className="hidden lg:block">
              <ProfileSidebar
                data={{
                  bio: userProfile.bio,
                  location: userProfile.location,
                  website: userProfile.website,
                  joinDate: userProfile.createdAt,
                  recentPhotos: [

        {activeTab === 'about' && (
          <div className="bg-white dark:bg-slate-900 rounded-2xl border border-[#E5E7EB] dark:border-slate-700 shadow-sm p-8">
            <h2 className="text-2xl font-bold text-[#111827] dark:text-gray-100 mb-6">{t('community.profileAboutTitle')}</h2>
